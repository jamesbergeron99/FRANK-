let audioCtx; let isSequenceActive = false; let nextToPlay = 0; let audioBuffers = {}; let recognition; let isRecording = false; let scriptMemory = {}; let feedbackHistory = []; function initAudio() { if (!audioCtx) { try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } catch (e) { console.error('AudioContext not supported:', e); } } if (audioCtx && audioCtx.state === 'suspended') { audioCtx.resume().then(() => console.log('AudioContext resumed')); } } function initSpeechRecognition() { try { recognition = new (window.SpeechRecognition || window.webkitSpeechRecognition || window.mozSpeechRecognition || window.msSpeechRecognition)(); recognition.continuous = false; recognition.interimResults = false; recognition.lang = 'en-US'; recognition.onstart = () => { isRecording = true; document.getElementById('micButton').classList.add('recording'); document.getElementById('micButton').textContent = '🎤 Listening...'; }; recognition.onresult = (event) => { let transcript = ''; for (let i = event.resultIndex; i < event.results.length; i++) { transcript += event.results[i][0].transcript; } const currentSection = document.querySelector('.section-toggle.active').onclick.toString().includes('feature') ? 'feature' : 'tv'; const contentId = currentSection === 'feature' ? 'scriptContent' : 'tvScriptContent'; document.getElementById(contentId).value += ' ' + transcript; }; recognition.onerror = (event) => { console.error('Speech recognition error:', event.error); }; recognition.onend = () => { isRecording = false; document.getElementById('micButton').classList.remove('recording'); document.getElementById('micButton').textContent = '🎤 Dictate'; }; } catch (e) { console.error('Speech Recognition not supported:', e); document.getElementById('micButton').style.opacity = '0.5'; document.getElementById('micButton').disabled = true; } } function toggleMic() { if (!recognition) initSpeechRecognition(); if (isRecording) { recognition.stop(); } else { recognition.start(); } } function toggleSection(section) { stopAudio(); fetch('/tv-greeting').then(r => r.text()).then(text => speak(text)); document.querySelectorAll('.section-toggle').forEach(btn => btn.classList.remove('active')); document.querySelectorAll('.input-section').forEach(sec => sec.style.display = 'none'); if (section === 'feature') { document.querySelector('[onclick="toggleSection(\'feature\')"]').classList.add('active'); document.getElementById('featureSection').style.display = 'block'; } else { document.querySelector('[onclick="toggleSection(\'tv\')"]').classList.add('active'); document.getElementById('tvSection').style.display = 'block'; } } function stopAudio() { isSequenceActive = false; nextToPlay = 0; audioBuffers = {}; if (audioCtx) { audioCtx.getDestination().context.state === 'running' && audioCtx.close(); audioCtx = null; } } function speak(text) { if (!text || text.trim().length === 0) return; initAudio(); isSequenceActive = true; nextToPlay = 0; const sections = text.split('\n\n').filter(s => s.trim()); const validSections = sections.filter(s => s.trim().length > 0); validSections.forEach((section, index) => { fetch('/tts', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ text: section }) }) .then(r => r.arrayBuffer()) .then(arrayBuffer => { if (isSequenceActive) { audioCtx.decodeAudioData(arrayBuffer, buffer => { audioBuffers[index] = buffer; playSequentially(); }, err => console.error('Decode error:', err)); } }) .catch(err => console.error('TTS error:', err)); }); } function playSequentially() { if (!isSequenceActive || !audioCtx) return; const keys = Object.keys(audioBuffers).map(Number).sort((a, b) => a - b); if (nextToPlay < keys.length && audioBuffers[keys[nextToPlay]]) { const source = audioCtx.createBufferSource(); source.buffer = audioBuffers[keys[nextToPlay]]; source.connect(audioCtx.destination); source.onended = () => { nextToPlay++; playSequentially(); }; source.start(0); } } function submitScript(type) { const isFea = type === 'feature'; const title = document.getElementById(isFea ? 'scriptTitle' : 'tvTitle').value || 'Untitled'; const content = document.getElementById(isFea ? 'scriptContent' : 'tvScriptContent').value; if (!content) { alert('Please enter script content'); return; } fetch('/analyze', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ scriptType: type, title: title, content: content, memory: scriptMemory }) }) .then(r => r.json()) .then(data => { scriptMemory = data.updatedMemory || scriptMemory; feedbackHistory.push(data.feedback); document.getElementById('feedback').textContent = data.feedback; speak(data.feedback); }) .catch(err => console.error('Error:', err));
